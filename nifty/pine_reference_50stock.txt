// % of `cur` vs an arbitrary `base`, na-safe.
f_pct(cur, base) => (not na(cur) and not na(base) and base != 0) ? (cur - base) / base * 100.0 : na

// ── PER-CHART-BAR % (the key to per-candle lines on ANY timeframe) ─────────────
// Evaluated INSIDE request.security at the CHART's OWN timeframe, so `close` updates
// every candle instead of returning one flat daily value. All anchors are captured
// from that same per-bar series via `var`s:
//   today's open (Intraday), prev day/week/month close (Daily/Weekly/Monthly),
//   today's 15:00 open (3PM), today's HH:MM open (Custom).
f_chart_pct() =>
    var float dopen = na   // today's open
    var float pdc   = na   // previous day's close
    var float pwc   = na   // previous week's close
    var float pmc   = na   // previous month's close
    var float a3pm  = na   // today's 15:00 open
    var float acus  = na   // today's HH:MM open
    bool nd = na(time[1]) or dayofmonth != dayofmonth[1] or month != month[1] or year != year[1]
    if nd
        dopen := open
        pdc   := close[1]
        a3pm  := na
        acus  := na
    if weekofyear != weekofyear[1]
        pwc := close[1]
    if month != month[1]
        pmc := close[1]
    int cur_min = hour * 60 + minute
    if na(a3pm) and cur_min >= 900                          // 15:00 = 3 PM
        a3pm := open
    if na(acus) and cur_min >= (anchor_hh * 60 + anchor_mm)
        acus := open
    float base = chg_basis == "Intraday %chg" ? dopen : chg_basis == "Weekly %chg" ? pwc : chg_basis == "Monthly %chg" ? pmc : chg_basis == "Intraday 3PM" ? a3pm : chg_basis == "Custom" ? acus : pdc
    f_pct(close, base)

// ONE request per symbol, at the CHART timeframe -> per-candle data, 40-call budget intact.
// Requesting the chart's OWN resolution means no timeframe mismatch (no memory blow-up);
// calc_bars_count=5000 caps history — enough to span week / most month boundaries so the
// Weekly/Monthly anchors resolve (very fine charts may leave Monthly na until enough history).
get_perf_tf(sym) =>
    request.security(sym, timeframe.period, f_chart_pct(), lookahead = barmerge.lookahead_off, gaps = barmerge.gaps_off, calc_bars_count = 5000)

// ==============================================================================
// EXTRA-11 DECODER — unpack the Helper's single packed plot, WITH slot-ID checking.
// Packed layout (14 digits): SS DDD III WWW MMM
//   SS  = slot id 1..11 (identifies WHICH stock this plot is)
//   DDD III WWW WWW = Daily/Intraday/Weekly/Monthly % offsets (mult 10/10/5/5)
// This lets the main verify each source is linked to the RIGHT Helper plot:
//   • embedded slot 0            -> unlinked (source still = chart close) -> na
//   • embedded slot != expected  -> MIS-LINKED (wrong plot) -> na + cell warning
//   • embedded slot == expected  -> decode the % for the selected Basis
// ==============================================================================
f_slot_of(packed) => na(packed) ? 0 : int((packed - (packed % 1000000000000)) / 1000000000000)   // 0 = unlinked, else 1..11

f_unpack_body(packed) =>
    float body  = packed % 1000000000000
    int   d_off = int(math.floor(body / 1000000000))
    int   i_off = int(math.floor((body % 1000000000) / 1000000))
    int   w_off = int(math.floor((body % 1000000) / 1000))
    int   m_off = int(body % 1000)
    chg_basis == "Intraday %chg" ? (i_off - 500) / 10.0 : chg_basis == "Weekly %chg" ? (w_off - 500) / 5.0 : chg_basis == "Monthly %chg" ? (m_off - 500) / 5.0 : (d_off - 500) / 10.0

// % for a slot — only when linked to the CORRECT plot and not in Custom mode.
f_xpct(packed, expect) => (f_slot_of(packed) == expect and chg_basis != "Custom" and chg_basis != "Intraday 3PM") ? f_unpack_body(packed) : na
// True when the source is linked to SOME Helper plot but the WRONG one.
f_mis(packed, expect)  => f_slot_of(packed) != 0 and f_slot_of(packed) != expect

// Constituent stock indices for a sector — DERIVED from the stock_sectors master
// array (no hardcoded lists). Returns indices in ascending (weight-descending) order.
// The extra-11 are included automatically; each decodes to na when its Helper source
// is unlinked (or in Custom mode).
f_get_sector_stocks(s_idx) =>
    var int[] stks = array.new_int(0)
    array.clear(stks)
    for i = 0 to array.size(stock_sectors) - 1
        if array.get(stock_sectors, i) == s_idx
            array.push(stks, i)
    stks

// Sector % = weight-weighted average of the constituents that HAVE data (na-aware,
// DYNAMIC denominator). Fully linked -> denominator = official sector weight (exact);
// unlinked extras drop out -> graceful fallback to the 39-stock behaviour.
get_sector_pct(s_idx) =>
    int[] stks = f_get_sector_stocks(s_idx)
    float wsum = 0.0
    float psum = 0.0
    for i = 0 to array.size(stks) - 1
        int idx = array.get(stks, i)
        float p = get_stock_pct(idx)
        if not na(p)
            float w = array.get(stock_weights, idx)
            wsum += w
            psum += w * p
    wsum > 0 ? psum / wsum : na

// ==============================================================================
// INDEX POINTS CONTRIBUTION ENGINE (Part 1: per-stock + formatter)
// ------------------------------------------------------------------------------
// points = NIFTY level × (weight% / 100) × (change% / 100)
// This is the standard index-attribution approximation. It is mathematically
// identical to the manual "share = idxLtp * weight * chgP" method (exact up to
// a negligible 2nd-order term). A positive value = the stock is pushing NIFTY
// UP by that many points; negative = pulling it DOWN.
//
// NOTE: get_stock_points is defined HERE (early) because f_build_tooltip below
// needs it. get_sector_points is defined later, after f_get_sector_stocks exists.
// ==============================================================================
get_stock_points(idx) =>
    float pct = get_stock_pct(idx)
    float wt  = array.get(stock_weights, idx)
    (na(pct) or na(nifty50_price)) ? na : nifty50_price * (wt / 100.0) * (pct / 100.0)

bool do_sort_stocks  = sort_mode == "ALL Performance %" or sort_mode == "ALL Points"
// Descending sort key for a stock (na sinks to the bottom).
f_stock_key(idx) => sort_by_points ? nz(get_stock_points(idx), -999999.0) : nz(get_stock_pct(idx), -999.0)

// Returns the pre-formatted points chunk (e.g. "+40.2 pt  ") or "" when the
// toggle is OFF / value is na -> guarantees identical output when disabled.
f_fmt_points(pts) =>
    (not show_points or na(pts)) ? "" : (pts > 0 ? "+" : "") + str.tostring(pts, "#.#") + " pt  "

// Formats an HHMM int (e.g. 915) as a padded clock string ("09:15") for the banner.
f_fmt_hhmm(v) =>
    int h = int(v / 100)
    int m = int(v % 100)
    (h < 10 ? "0" : "") + str.tostring(h) + ":" + (m < 10 ? "0" : "") + str.tostring(m)

calc_weighted_sentiment() =>
    weighted_positive = 0.0
    weighted_negative = 0.0
    for i = 0 to 14
        weight = array.get(sector_weights, i)
        pct = get_sector_pct(i)
        if not na(pct) and weight > 0
            weighted_contribution = (weight / 100.0) * pct
            if pct > 0
                weighted_positive += weighted_contribution
            else if pct < 0
                weighted_negative += math.abs(weighted_contribution)
    [math.max(0, math.min(100, (weighted_positive / (weighted_positive + weighted_negative) * 100))), weighted_positive, weighted_negative]

// ==============================================================================
// TOOLTIP DATA GENERATORS   (f_get_sector_stocks is defined earlier, by the sector engine)
// ==============================================================================
f_build_tooltip(s_idx, s_mode) =>
    int[] stks = f_get_sector_stocks(s_idx)
    int sz = array.size(stks)

    // Sort array identically to the selected view mode (Fixed sz check)
    if sz > 1 and do_sort_stocks
        for x = 0 to sz - 2
            for y = 0 to sz - 2 - x
                int a = array.get(stks, y)
                int b = array.get(stks, y + 1)
                if f_stock_key(a) < f_stock_key(b)
                    array.set(stks, y, b)
                    array.set(stks, y + 1, a)

    string tt = "--- " + str.upper(get_short_name(s_idx)) + " STOCKS ---\n"
    for i = 0 to sz - 1
        int idx = array.get(stks, i)
        float pct = get_stock_pct(idx)
        string name = array.get(stock_names, idx)
        string dir = na(pct) ? "" : (pct > 0 ? "▲ " : (pct < 0 ? "▼ " : ""))
        string val_str = na(pct) ? "--" : str.tostring(pct, "#.##") + "%"
        string pts_str = f_fmt_points(get_stock_points(idx))
        tt += name + ": " + pts_str + dir + val_str + (i < sz - 1 ? "\n" : "")
    tt

f_get_sorted_sector_stocks(s_idx) =>
    int[] stks = f_get_sector_stocks(s_idx)
    int sz = array.size(stks)

    // (Fixed sz check)
    if sz > 1 and do_sort_stocks
        for x = 0 to sz - 2
            for y = 0 to sz - 2 - x
                int a = array.get(stks, y)
                int b = array.get(stks, y + 1)
                if f_stock_key(a) < f_stock_key(b)
                    array.set(stks, y, b)
                    array.set(stks, y + 1, a)
    stks

// ==============================================================================
// INDEX POINTS CONTRIBUTION ENGINE (Part 2: per-sector)
// ------------------------------------------------------------------------------
// Two selectable bases (see 'Sector Points Basis' input):
//   • "Official NSE Weight" -> NIFTY level × official_sector_weight% × sector%.
//       Scales the tracked stocks up to the full NSE sector weight. Best estimate
//       of true impact; all sectors sum ≈ to NIFTY's real point move.
//   • "Tracked Stocks Only" -> sum of the tracked constituents' points. Exact &
//       fully additive (header = sum of its shown stocks) but ~5% understated.
// ==============================================================================
get_sector_points(s_idx) =>
    if sector_pts_basis == "Official NSE Weight"
        float pct = get_sector_pct(s_idx)
        (na(pct) or na(nifty50_price)) ? na : nifty50_price * (array.get(sector_weights, s_idx) / 100.0) * (pct / 100.0)
    else
        int[] stks = f_get_sector_stocks(s_idx)
        float total = 0.0
        bool  any   = false
        for i = 0 to array.size(stks) - 1
            float p = get_stock_points(array.get(stks, i))
            if not na(p)
                total += p
                any := true
        any ? total : na

// Descending sort key for a sector (na sinks to the bottom).
f_sector_key(s_idx) => sort_by_points ? nz(get_sector_points(s_idx), -999999.0) : nz(get_sector_pct(s_idx), -999.0)

// Weighted % change across all 15 sectors — returns [positive%, negative%]
// (neg is a positive magnitude). These ARE the actual weighted %-change contributions
// (same numbers as the header's "Weighted Chg" +/-), NOT the gauge's share-of-move.
f_wbreadth() =>
    float pw = 0.0
    float nw = 0.0
    for i = 0 to 14
        float sp = get_sector_pct(i)
        float w  = array.get(sector_weights, i)
        if not na(sp)
            if sp > 0
                pw += w / 100.0 * sp
            else if sp < 0
                nw += w / 100.0 * (-sp)
    [pw, nw]

// Horizontal fill-bar for the Weighted-Chg gauge. Fixed 12-cell width: `share`
// (0-100) worth of solid "█" (fill) then faint "░" (track) — so the bar's full
// extent is always visible and share=100 fills it entirely. Track "░" over the
// gauge-BG cell reads as the unfilled portion.
f_gauge_bar(share) =>
    int w = 12
    int n = math.max(0, math.min(w, int(math.round(nz(share) / 100.0 * w))))
    string s = ""
    for k = 0 to w - 1
        s += k < n ? "█" : "░"
    s

// ==============================================================================
// STOCK CELL BUILDERS — one place that builds every stock cell's text & colours,
// including the ⚠ WRONG-LINK warning for a mis-linked extra stock.

f_stock_cell_text(idx) =>

    hdr + body

f_stock_cell_bg(idx) => get_stock_mislinked(idx) ? color.new(#ff9800, 0) : f_get_bg(get_stock_pct(idx), false)

// Generic sector block: header (5-wide) then the sector's stocks wrapped 5 per row,
// each row's cells sharing the 5 columns as evenly as possible. Handles any count
// (1 stock -> full-width; 11 stocks like Finance -> rows of 5, 5, 1).
f_render_sector(table tbl, int r_start, int c_offset, int s_idx, string txt_size) =>
    f_render_sector_header(tbl, r_start, c_offset, s_idx, txt_size)
    int[] stks = f_get_sorted_sector_stocks(s_idx)
    int n = array.size(stks)
    int r = r_start + 1
    int i = 0
    while i < n
        int k    = int(math.min(5, n - i))   // cells to place in this row
        int base = int(5 / k)                // even column spans across the 5-wide block
        int rem  = 5 - base * k
        int col  = 0
        for j = 0 to k - 1
            int span = base + (j < rem ? 1 : 0)
            f_render_stock(tbl, c_offset + col, r, array.get(stks, i + j), txt_size)
            if span > 1
                table.merge_cells(tbl, c_offset + col, r, c_offset + col + span - 1, r)
            col := col + span
        r := r + 1
        i := i + k
    r - r_start   // rows consumed (header + stock rows)

            table.cell(main_tbl, 3, 3, neg_contrib_str, text_color = wg_neg_col, bgcolor = color.new(color.white, 100), text_size = selected_text_size, text_halign = text.align_left)
            table.cell(main_tbl, 4, 3, f_gauge_bar(neg_share), text_color = wg_neg_col, bgcolor = wg_gauge_bg, text_size = selected_text_size, text_halign = text.align_left)
        else
            table.cell(main_tbl, 4, 2, neg_contrib_str, text_color = wg_neg_col, bgcolor = color.new(color.white, 100), text_size = selected_text_size, text_halign = text.align_left)

    // ── Row 3: anchor banner for non-Daily bases; "Daily %chg" leaves this row blank ──
    // When the gauge is ON, row 3's right side holds the neg% + neg gauge, so the banner
    // and the blank spacer only occupy the LEFT columns (0..wg_left_end).
    int wg_left_end = view_mode == "Sector Grouped View" ? 6 : 2
    int row3_end    = show_wchg_gauge ? wg_left_end : table_cols - 1
    string anchor_label = chg_basis == "Intraday %chg" ? "Today's Open" : chg_basis == "Intraday 3PM" ? "the 15:00 (3 PM) candle Open" : chg_basis == "Custom" ? "the " + f_fmt_hhmm(anchor_hhmm) + " candle Open" : ""
    if anchor_label != ""
        table.cell(main_tbl, 0, 3, "⚓ Prices anchored to " + anchor_label, text_color = #1976d2, bgcolor = color.new(color.white, 100), text_size = selected_text_size, text_halign = text.align_center)
        table.merge_cells(main_tbl, 0, 3, row3_end, 3)

    if view_mode == "Sector Grouped View"
        if anchor_label == ""
            for col = 0 to row3_end
                table.cell(main_tbl, col, 3, " ", bgcolor = color.new(color.white, 100), text_size = size.tiny)

        var int[] sector_order_gv = array.new_int(15)
        for k = 0 to 14
            array.set(sector_order_gv, k, k)

        if do_sort_sectors
            for x = 0 to 13
                for y = 0 to 13 - x
                    s_a = array.get(sector_order_gv, y)
                    s_b = array.get(sector_order_gv, y + 1)
                    if f_sector_key(s_a) < f_sector_key(s_b)
                        array.set(sector_order_gv, y, s_b)
                        array.set(sector_order_gv, y + 1, s_a)

        // Independent row trackers for left and right columns
        int r_left = 4
        int r_right = 4

        for i = 0 to 14
            int s_idx = array.get(sector_order_gv, i)

            // Greedily place the next sector in the column that has fewer rows used
            if r_left <= r_right
                r_left += f_render_sector(main_tbl, r_left, 0, s_idx, selected_text_size)
            else
                r_right += f_render_sector(main_tbl, r_right, 6, s_idx, selected_text_size)

        int max_rows = math.max(r_left, r_right)

        // Generate visual border buffer splitting left and right table blocks
        for r_fill = 4 to max_rows - 1
            table.cell(main_tbl, 5, r_fill, " ", bgcolor = color.new(color.white, 100), text_size = size.tiny)

    else // Standard Multi-Matrix Mode
        if (show_grid or is_combined or show_wchg_gauge) and anchor_label == ""
            for col = 0 to row3_end
                table.cell(main_tbl, col, 3, " ", bgcolor = color.new(color.white, 100), text_size = size.tiny)

        var int[] sector_order = array.new_int(15)
        var int[] stock_order = array.new_int(50)

        for k = 0 to 14
            array.set(sector_order, k, k)
        for k = 0 to stock_limit - 1
            array.set(stock_order, k, k)

        if do_sort_sectors
            for x = 0 to 13
                for y = 0 to 13 - x
                    s_a = array.get(sector_order, y)
                    s_b = array.get(sector_order, y + 1)
                    if f_sector_key(s_a) < f_sector_key(s_b)
                        array.set(sector_order, y, s_b)
                        array.set(sector_order, y + 1, s_a)

        if do_sort_stocks
            int sort_stk_limit = stock_limit
            for x = 0 to sort_stk_limit - 2
                for y = 0 to sort_stk_limit - 2 - x
                    stk_a = array.get(stock_order, y)
                    stk_b = array.get(stock_order, y + 1)
                    if f_stock_key(stk_a) < f_stock_key(stk_b)
                        array.set(stock_order, y, stk_b)
                        array.set(stock_order, y + 1, stk_a)

        if show_grid
            for i = 0 to 14
                int target_idx = array.get(sector_order, i)
                pct = get_sector_pct(target_idx)
                float pts = get_sector_points(target_idx)
                grid_row = math.floor(i / 5) + 4
                grid_col = i % 5

                string tt_text = f_build_tooltip(target_idx, sort_mode)

                cell_text = get_short_name(target_idx) + " (" + str.tostring(array.get(sector_weights, target_idx), "#.#") + "%)\n\n" + f_fmt_points(pts) + (na(pct) ? "" : (pct > 0 ? "▲ " : (pct < 0 ? "▼ " : ""))) + (na(pct) ? "--" : str.tostring(pct, "#.##") + "%")
                table.cell(main_tbl, grid_col, grid_row, cell_text, text_color = f_get_text_col(pct, true), bgcolor = f_get_bg(pct, true), text_size = selected_text_size, text_halign = text.align_center, tooltip = tt_text)

        int stock_start_row = show_grid ? 8 : 4

        if is_combined
            if show_grid
                for col = 0 to 4
                    table.cell(main_tbl, col, 7, " ", bgcolor = color.new(color.white, 100), text_size = selected_text_size)

            for i = 0 to stock_limit - 1
                int target_idx = array.get(stock_order, i)
                grid_row = math.floor(i / 5) + stock_start_row
                grid_col = i % 5
                table.cell(main_tbl, grid_col, grid_row, f_stock_cell_text(target_idx), text_color = f_stock_cell_tc(target_idx), bgcolor = f_stock_cell_bg(target_idx), text_size = selected_text_size, text_halign = text.align_center)

        else if show_stocks and stocks_layout == "Separate Standalone Table"
            for i = 0 to stock_limit - 1
                int target_idx = array.get(stock_order, i)
                grid_row = math.floor(i / 5)
                grid_col = i % 5
                table.cell(stock_tbl, grid_col, grid_row, f_stock_cell_text(target_idx), text_color = f_stock_cell_tc(target_idx), bgcolor = f_stock_cell_bg(target_idx), text_size = selected_text_size, text_halign = text.align_center)

// ==============================================================================
// WEIGHTED-CHANGE EXPORTS (for the companion "Nifty50 Weighted Chg" pane indicator)
// The weighted % CHANGE (positive above 0, negative below 0) — the same numbers as
// the header's "Weighted Chg" +/-, NOT the gauge's share-of-move. Computed every bar.
// Plotted display.none — invisible on THIS overlay, but linkable as an input.source.
// In the companion indicator, link "Weighted Chg +" (green) and "Weighted Chg -" (red).
// ==============================================================================
[wb_pw, wb_nw] = f_wbreadth()
float wchg_pos_export = wb_pw                          // weighted POSITIVE % change (above 0)
float wchg_neg_export = use_abs_lines ? wb_nw : -wb_nw // negative %: absolute (above 0, comparable) or mirrored below 0
plot(wchg_pos_export, title = "Weighted Chg +", color = color.green, display = display.none)
plot(wchg_neg_export, title = "Weighted Chg -", color = color.red,   display = display.none)
